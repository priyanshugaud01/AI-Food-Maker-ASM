📄 Supabase SQL Table Schemas & RLS

Run this SQL script in your Supabase SQL Editor to provision the exact tables with Row Level Security (RLS) configured:

-- TasteAI Supabase Schema & Row Level Security -- 1. Profiles Table CREATE TABLE IF NOT EXISTS public.profiles ( id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY, name TEXT, email TEXT, age INTEGER, gender TEXT, dietary_preference TEXT, favorite_cuisine TEXT, nutrition_goal TEXT, created_at TIMESTAMPTZ DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL ); ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users can manage own profile" ON public.profiles FOR ALL USING (auth.uid() = id); -- 2. Recipes Table CREATE TABLE IF NOT EXISTS public.recipes ( id TEXT PRIMARY KEY, user_id UUID REFERENCES auth.users ON DELETE CASCADE, recipe_name TEXT NOT NULL, description TEXT, ingredients JSONB, instructions JSONB, cuisine TEXT, meal_type TEXT, dietary_type TEXT, calories INTEGER, protein NUMERIC, carbohydrates NUMERIC, fat NUMERIC, fiber NUMERIC, cooking_time TEXT, difficulty TEXT, servings INTEGER, created_at TIMESTAMPTZ DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL ); ALTER TABLE public.recipes ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users can manage own recipes" ON public.recipes FOR ALL USING (auth.uid() = user_id); -- 3. Favorites Table CREATE TABLE IF NOT EXISTS public.favorites ( id BIGSERIAL PRIMARY KEY, user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL, recipe_id TEXT NOT NULL, created_at TIMESTAMPTZ DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL, UNIQUE(user_id, recipe_id) ); ALTER TABLE public.favorites ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users can manage own favorites" ON public.favorites FOR ALL USING (auth.uid() = user_id);
